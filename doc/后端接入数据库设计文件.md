-- 用户表
CREATE TABLE users (
  id VARCHAR(32) PRIMARY KEY,
  name VARCHAR(50) NOT NULL,
  email VARCHAR(100) UNIQUE,
  role ENUM('manager', 'employee') DEFAULT 'employee',
  department VARCHAR(50),
  avatar_url TEXT,
  feishu_open_id VARCHAR(50),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 任务表
CREATE TABLE tasks (
  id VARCHAR(32) PRIMARY KEY,
  title VARCHAR(200) NOT NULL,
  description TEXT,
  project_id VARCHAR(32),
  assignee_id VARCHAR(32),
  priority ENUM('urgent', 'high', 'medium', 'low') DEFAULT 'medium',
  status ENUM('not-started', 'in-progress', 'pending-review', 'completed', 'overdue') DEFAULT 'not-started',
  progress INT DEFAULT 0,
  due_date DATE,
  start_date DATE,
  created_by VARCHAR(32),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 项目表
CREATE TABLE projects (
  id VARCHAR(32) PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  health_score INT DEFAULT 100,
  progress INT DEFAULT 0,
  status ENUM('active', 'completed', 'at-risk') DEFAULT 'active'
);

-- 日报表
CREATE TABLE daily_reports (
  id VARCHAR(32) PRIMARY KEY,
  employee_id VARCHAR(32),
  date DATE NOT NULL,
  completed_tasks JSON,
  tomorrow_plan JSON,
  blockers TEXT,
  support_needed TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 求助表
CREATE TABLE help_requests (
  id VARCHAR(32) PRIMARY KEY,
  employee_id VARCHAR(32),
  task_id VARCHAR(32),
  reason TEXT,
  status ENUM('pending', 'resolved') DEFAULT 'pending',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 飞书配置表
CREATE TABLE feishu_configs (
  id INT PRIMARY KEY AUTO_INCREMENT,
  app_id VARCHAR(100),
  app_secret VARCHAR(200),
  webhook_url TEXT,
  connected BOOLEAN DEFAULT FALSE,
  connected_at TIMESTAMP
);
